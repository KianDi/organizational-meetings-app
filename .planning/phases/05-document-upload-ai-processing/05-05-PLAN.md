### 1. Load Tasks When Displaying Meeting
**What:** Fetch tasks alongside meeting data for display
**Files:** `MeetingManager/State/MeetingState.swift`
**Why:** Tasks stored separately, need to load for UI

**Changes:**
```swift
@Observable
final class MeetingState {
    // Add task service dependency (already added in 05-04)
    private let taskService: TaskService

    @MainActor
    func loadMeetingWithTasks(meetingId: UUID) async throws -> Meeting {
        // Fetch meeting
        var meeting = try await meetingService.fetchMeeting(meetingId: meetingId)

        // Fetch associated tasks
        let tasks = try await taskService.fetchTasksForMeeting(meetingId: meetingId)
        meeting.tasks = tasks

        // Update local cache
        if let index = meetings.firstIndex(where: { $0.id == meetingId }) {
            meetings[index] = meeting
        }

        return meeting
    }

    @MainActor
    func loadMeetings(organizationId: UUID) async throws {
        isLoading = true
        defer { isLoading = false }

        // Fetch meetings
        let fetchedMeetings = try await meetingService.fetchMeetingsForOrganization(
            organizationId: organizationId
        )

        // Optionally fetch tasks for each meeting
        // For performance, might skip this and load tasks on-demand per meeting
        meetings = fetchedMeetings
    }
}
```

```swift
import SwiftUI

struct TaskRowView: View {
    // MARK: - Properties

    let task: MeetingTask

    // For displaying assignee name (passed from parent)
    var assigneeName: String?

    // MARK: - Body

    var body: some View {
        HStack(alignment: .top, spacing: 12) {
            // Completion indicator (read-only in this view)
            Image(systemName: task.isCompleted ? "checkmark.circle.fill" : "circle")
                .foregroundStyle(task.isCompleted ? .green : .gray)
                .font(.title3)

            // Task content
            VStack(alignment: .leading, spacing: 4) {
                // Title
                Text(task.title)
                    .font(.body)
                    .foregroundStyle(.primary)
                    .strikethrough(task.isCompleted)

                // Metadata row
                HStack(spacing: 12) {
                    // Assignee
                    if let assigneeName {
                        Label(assigneeName, systemImage: "person.fill")
                            .font(.caption)
                            .foregroundStyle(.secondary)
                    } else {
                        Label("Unassigned", systemImage: "person.fill.questionmark")
                            .font(.caption)
                            .foregroundStyle(.orange)
                    }

                    // Due date
                    if let dueDate = task.dueDate {
                        Label(formatDate(dueDate), systemImage: "calendar")
                            .font(.caption)
                            .foregroundStyle(isOverdue(dueDate) ? .red : .secondary)
                    }
                }
            }

            Spacer()
        }
        .padding()
        .background(Color(.tertiarySystemBackground))
        .cornerRadius(8)
    }

    // MARK: - Helpers

    private func formatDate(_ date: Date) -> String {
        let calendar = Calendar.current
        let now = Date()

        if calendar.isDateInToday(date) {
            return "Today"
        } else if calendar.isDateInTomorrow(date) {
            return "Tomorrow"
        } else if calendar.isDate(date, equalTo: now, toGranularity: .weekOfYear) {
            // This week - show day name
            let formatter = DateFormatter()
            formatter.dateFormat = "EEEE"
            return formatter.string(from: date)
        } else {
            // Future - show full date
            let formatter = DateFormatter()
            formatter.dateStyle = .medium
            return formatter.string(from: date)
        }
    }

    private func isOverdue(_ date: Date) -> Bool {
        return date < Date() && !task.isCompleted
    }
}
```

### 4. Fetch and Display Assignee Names
**What:** Load User objects for task assignees and display names
**Files:** `MeetingManager/Views/Meeting/MeetingDetailView.swift`
**Why:** TaskRowView needs assignee names, not just UUIDs

**Changes:**
```swift
struct MeetingDetailView: View {
    @Environment(MeetingState.self) private var meetingState
    @Environment(OrganizationState.self) private var organizationState

    // Cache of assignee names by userId
    @State private var assigneeNames: [UUID: String] = [:]

    var body: some View {
        ScrollView {
            VStack(alignment: .leading, spacing: 20) {
                // ... sections ...

                if let tasks = currentMeeting.tasks, !tasks.isEmpty {
                    tasksSection(tasks)
                }
            }
        }
        .task {
            await loadAssigneeNames()
        }
    }

    @ViewBuilder
    private func tasksSection(_ tasks: [MeetingTask]) -> some View {
        VStack(alignment: .leading, spacing: 12) {
            HStack(spacing: 8) {
                Image(systemName: "checkmark.circle")
                    .foregroundStyle(.green)
                Text("Action Items")
                    .font(.title3)
                    .fontWeight(.semibold)
                Text("(\(tasks.count))")
                    .font(.subheadline)
                    .foregroundStyle(.secondary)
            }

            VStack(spacing: 8) {
                ForEach(tasks) { task in
                    TaskRowView(
                        task: task,
                        assigneeName: task.assigneeId.flatMap { assigneeNames[$0] }
                    )
                }
            }
        }
        .padding(.vertical, 8)
    }

    private func loadAssigneeNames() async {
        guard let tasks = currentMeeting.tasks else { return }

        // Get unique assignee IDs
        let assigneeIds = Set(tasks.compactMap { $0.assigneeId })

        // Fetch members if not already loaded
        if organizationState.members.isEmpty {
            do {
                try await organizationState.loadMembers(organizationId: currentMeeting.organizationId)
            } catch {
                print("Failed to load members: \(error)")
                return
            }
        }

        // Build name mapping
        var names: [UUID: String] = [:]
        for id in assigneeIds {
            if let member = organizationState.members.first(where: { $0.id == id }) {
                names[id] = member.name
            }
        }

        await MainActor.run {
            assigneeNames = names
        }
    }
}
```

### 7. Add Share Summary Feature
**What:** Allow users to share meeting summary via share sheet
**Files:** `MeetingManager/Views/Meeting/MeetingDetailView.swift`
**Why:** Users may want to distribute summary to non-members

**Changes:**
```swift
struct MeetingDetailView: View {
    @State private var showShareSheet = false
    @State private var shareText = ""

    var body: some View {
        ScrollView {
            // ... content ...
        }
        .navigationTitle(currentMeeting.title)
        .toolbar {
            // Add share button if summary exists
            if currentMeeting.summary != nil {
                ToolbarItem(placement: .topBarTrailing) {
                    Button {
                        prepareShareText()
                        showShareSheet = true
                    } label: {
                        Image(systemName: "square.and.arrow.up")
                    }
                }
            }
        }
        .sheet(isPresented: $showShareSheet) {
            if #available(iOS 16.0, *) {
                ShareSheet(items: [shareText])
            }
        }
    }

    private func prepareShareText() {
        var text = """
        \(currentMeeting.title)
        \(formatDate(currentMeeting.scheduledAt))

        SUMMARY
        \(currentMeeting.summary ?? "")
        """

        // Add tasks if present
        if let tasks = currentMeeting.tasks, !tasks.isEmpty {
            text += "\n\nACTION ITEMS\n"
            for (index, task) in tasks.enumerated() {
                text += "\n\(index + 1). \(task.title)"
                if let assigneeId = task.assigneeId,
                   let name = assigneeNames[assigneeId] {
                    text += " - \(name)"
                }
                if let dueDate = task.dueDate {
                    text += " (Due: \(formatDate(dueDate)))"
                }
            }
        }

        text += "\n\nGenerated with MeetingManager"
        shareText = text
    }
}

// Share sheet wrapper (reuse from OrganizationDetailView pattern)

    func makeUIViewController(context: Context) -> UIActivityViewController {
        UIActivityViewController(activityItems: items, applicationActivities: nil)
    }

    func updateUIViewController(_ uiViewController: UIActivityViewController, context: Context) {}
}
```

### 8. Polish Processing State UI
**What:** Improve progress indicator with estimated time remaining
**Files:** `MeetingManager/Views/Meeting/MeetingDetailView.swift`
**Why:** Better user feedback during long operations

**Changes:**
```swift
struct MeetingDetailView: View {
    @ViewBuilder
    private var processingStatusView: some View {
        HStack(spacing: 12) {
            if meetingState.processingState.isProcessing {
                ProgressView()
                    .progressViewStyle(CircularProgressViewStyle())

                VStack(alignment: .leading, spacing: 2) {
                    Text(meetingState.processingState.displayText)
                        .font(.subheadline)
                        .foregroundStyle(.primary)

                    Text(estimatedTimeRemaining)
                        .font(.caption)
                        .foregroundStyle(.secondary)
                }
            } else if meetingState.processingState == .completed {
                Image(systemName: "checkmark.circle.fill")
                    .foregroundStyle(.green)
                    .font(.title3)

                Text("Processing complete")
                    .font(.subheadline)
            }
        }
        .padding()
        .frame(maxWidth: .infinity, alignment: .leading)
        .background(Color(.secondarySystemBackground))
        .cornerRadius(12)
    }

    private var estimatedTimeRemaining: String {
        switch meetingState.processingState {
        case .uploadingDocument, .parsingDocument:
            return "~5 seconds"
        case .generatingSummary:
            return "~10 seconds"
        case .extractingTasks:
            return "~5 seconds"
        default:
            return ""
        }
    }
}
```

## Notes

**Decisions:**
- Summary displayed before tasks (priority order)
- Task completion shown as read-only in meeting detail (edit in Phase 6)
- Assignee names loaded separately (not in Meeting model)
- Share feature uses plain text format (not PDF, easier to implement)
- Empty state only shown when no document uploaded (not when processing fails)

**UI/UX considerations:**
- Summary uses generous line spacing for readability
- Tasks use cards with subtle background for separation
- Processing estimates help set expectations (5-10 seconds)
- Share button only appears after summary generated
- Color coding: blue (AI/summary), green (tasks), red (overdue)

**Performance:**
- Assignee names cached in @State (no repeated fetches)
- Tasks loaded on-demand per meeting (not in bulk list)
- Pull-to-refresh refreshes current meeting only
- No real-time updates (user manually refreshes)

**Accessibility:**
- All icons have semantic meaning reinforced by text
- Color indicators supplemented with icons (not color-only)
- Task completion uses checkmark icon (not just color)
- Share sheet uses native iOS accessibility

**Future considerations (not in this plan):**
- Task completion toggle in meeting detail (Phase 6)
- Inline task editing (change assignee, due date)
- Summary regeneration button (separate from document re-upload)
- Export to PDF with formatting
- Real-time updates via WebSocket/polling
- Summary read time estimate ("3 min read")
- Collapsible sections for long task lists
