```sql
-- Create tasks table
CREATE TABLE IF NOT EXISTS tasks (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    meeting_id UUID NOT NULL REFERENCES meetings(id) ON DELETE CASCADE,
    organization_id UUID NOT NULL REFERENCES organizations(id) ON DELETE CASCADE,
    title TEXT NOT NULL,
    assignee_id UUID REFERENCES auth.users(id),
    due_date TIMESTAMPTZ,
    is_completed BOOLEAN NOT NULL DEFAULT false,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    extracted_from TEXT,  -- Source text where task was mentioned
    priority TEXT,        -- 'high', 'medium', 'low'
    CONSTRAINT valid_priority CHECK (priority IS NULL OR priority IN ('high', 'medium', 'low'))
);

-- Create index for common queries
CREATE INDEX idx_tasks_organization ON tasks(organization_id);
CREATE INDEX idx_tasks_assignee ON tasks(assignee_id);
CREATE INDEX idx_tasks_meeting ON tasks(meeting_id);
CREATE INDEX idx_tasks_due_date ON tasks(due_date) WHERE due_date IS NOT NULL;

-- RLS policies: members can read all org tasks, only assignee can update completion
ALTER TABLE tasks ENABLE ROW LEVEL SECURITY;

-- Members can view all tasks in their organization
CREATE POLICY tasks_select_policy ON tasks
    FOR SELECT
    USING (
        organization_id IN (
            SELECT id FROM organizations
            WHERE auth.uid() = ANY(member_ids)
        )
    );

-- Organizers (meeting creators) can insert tasks
CREATE POLICY tasks_insert_policy ON tasks
    FOR INSERT
    WITH CHECK (
        meeting_id IN (
            SELECT id FROM meetings
            WHERE created_by_id = auth.uid()
        )
    );

-- Task assignees can update their own task completion status
CREATE POLICY tasks_update_policy ON tasks
    FOR UPDATE
    USING (assignee_id = auth.uid())
    WITH CHECK (assignee_id = auth.uid());

-- Meeting creators can delete tasks from their meetings
CREATE POLICY tasks_delete_policy ON tasks
    FOR DELETE
    USING (
        meeting_id IN (
            SELECT id FROM meetings
            WHERE created_by_id = auth.uid()
        )
    );
```

```swift
// Update ExtractedTaskData struct to match schema
struct ExtractedTaskData: Codable {
    let title: String
    let assigneeName: String?
    let dueDate: String?  // ISO8601 or natural language
    let priority: String  // "high", "medium", "low"
    let context: String?  // Source text
}

// Define schema for structured output
struct TaskExtractionSchema: Codable {
    let tasks: [ExtractedTaskData]
}

func extractTasks(from documentText: String, organizationMembers: [User]) async throws -> [ExtractedTaskData] {
    guard AnthropicConfig.isConfigured else {
        throw AnthropicError.notConfigured
    }

    // Build member context for better name matching
    let memberContext = organizationMembers
        .map { "\($0.name) (email: \($0.email))" }
        .joined(separator: "\n")

    return try await callWithRetry {
        let request = MessageRequest(
            model: .claude35Sonnet,
            maxTokens: AnthropicConfig.maxTokens,
            messages: [
                Message(role: .user, content: [
                    .text("""
                    Extract all action items and tasks from this meeting document.

                    Organization members:
                    \(memberContext)

                    Document:
                    \(documentText)

                    For each task, extract:
                    - title: Clear, actionable task description
                    - assigneeName: Person assigned (match to members list above, or null if unclear)
                    - dueDate: Deadline in ISO8601 format if mentioned (YYYY-MM-DD), or null
                    - priority: "high", "medium", or "low" based on urgency language
                    - context: Original text snippet where task was mentioned

                    Return JSON with a "tasks" array.
                    """)
                ])
            ],
            system: [
                .text("""
                You are a task extraction specialist. Extract action items from meeting documents.
                Match assignee names to the provided member list, handling typos and nicknames.
                If no assignee mentioned, leave assigneeName as null.
                If no due date mentioned, leave dueDate as null.
                Infer priority from urgency language (urgent/ASAP = high, normal = medium, backlog/someday = low).
                """),
                .cachingEphemeral  // Cache system prompt
            ],
            // Use structured outputs for guaranteed schema
            responseFormat: .json(schema: TaskExtractionSchema.jsonSchema)
        )

        let response = try await self.client.message(request)

        guard let textContent = response.content.first(where: {
            if case .text = $0 { return true }
            return false
        }) else {
            throw AnthropicError.parsingError
        }

        if case .text(let jsonString) = textContent {
            let data = jsonString.data(using: .utf8)!
            let schema = try JSONDecoder().decode(TaskExtractionSchema.self, from: data)
            return schema.tasks
        } else {
            throw AnthropicError.parsingError
        }
    }
}
```

### 4. Implement Assignee Name Matching Logic
**What:** Fuzzy match extracted names to organization members
**Files:** `MeetingManager/Utilities/NameMatcher.swift`
**Why:** Handle typos, nicknames, partial names in meeting documents

**Implementation:**
```swift
import Foundation

struct NameMatcher {
    /// Match an extracted name to a list of users
    /// Returns userId if confident match found, nil otherwise
    static func matchAssignee(
        name: String,
        candidates: [User]
    ) -> UUID? {
        let normalizedName = name.lowercased().trimmingCharacters(in: .whitespacesAndNewlines)

        // 1. Exact match on full name
        if let match = candidates.first(where: { $0.name.lowercased() == normalizedName }) {
            return match.id
        }

        // 2. Match on email prefix (before @)
        let emailPrefix = normalizedName.components(separatedBy: "@").first ?? normalizedName
        if let match = candidates.first(where: {
            $0.email.lowercased().hasPrefix(emailPrefix)
        }) {
            return match.id
        }

        // 3. Partial match on name components (first or last name)
        let nameComponents = normalizedName.split(separator: " ")
        if !nameComponents.isEmpty {
            for candidate in candidates {
                let candidateComponents = candidate.name.lowercased().split(separator: " ")
                // Check if any component matches
                for component in nameComponents {
                    if candidateComponents.contains(where: { $0.hasPrefix(component) || component.hasPrefix($0) }) {
                        return candidate.id
                    }
                }
            }
        }

        // 4. No confident match - return nil
        return nil
    }

    /// Parse natural language due date to Date
    /// Returns nil if can't parse confidently
    static func parseDate(from text: String) -> Date? {
        let normalized = text.lowercased().trimmingCharacters(in: .whitespacesAndNewlines)

        // Try ISO8601 first
        let iso8601Formatter = ISO8601DateFormatter()
        if let date = iso8601Formatter.date(from: text) {
            return date
        }

        // Handle common patterns
        let calendar = Calendar.current
        let now = Date()

        // "tomorrow"
        if normalized.contains("tomorrow") {
            return calendar.date(byAdding: .day, value: 1, to: now)
        }

        // "next week"
        if normalized.contains("next week") {
            return calendar.date(byAdding: .weekOfYear, value: 1, to: now)
        }

        // "friday", "monday", etc.
        let weekdays = ["monday", "tuesday", "wednesday", "thursday", "friday", "saturday", "sunday"]
        for (index, weekday) in weekdays.enumerated() {
            if normalized.contains(weekday) {
                // Find next occurrence of this weekday
                let targetWeekday = index + 1
                var components = calendar.dateComponents([.yearForWeekOfYear, .weekOfYear], from: now)
                components.weekday = targetWeekday
                if let date = calendar.date(from: components), date > now {
                    return date
                } else {
                    // Next week's occurrence
                    components.weekOfYear! += 1
                    return calendar.date(from: components)
                }
            }
        }

        // Can't parse - return nil
        return nil
    }
}
```

```swift
@Observable
final class MeetingState {
    // Add dependency
    private let taskService: TaskService
    private let organizationService: OrganizationService

    init(
        meetingService: MeetingService = MeetingService(),
        aiService: AIService = AIService(),
        taskService: TaskService = TaskService(),
        organizationService: OrganizationService = OrganizationService()
    ) {
        self.meetingService = meetingService
        self.aiService = aiService
        self.taskService = taskService
        self.organizationService = organizationService
    }

    @MainActor
    func processDocument(
        meetingId: UUID,
        documentUrl: URL,
        organizationId: UUID
    ) async {
        // ... existing upload, parse, summary logic ...

        do {
            // ... summary generation ...

            // Step 5: Extract tasks
            processingState = .extractingTasks

            // Fetch organization members for name matching
            let members = try await organizationService.fetchMembers(organizationId: organizationId)

            let extractedTasks = try await aiService.extractTasks(
                from: documentText,
                organizationMembers: members
            )

            // Convert extracted data to MeetingTask models
            let tasks = extractedTasks.map { extracted -> MeetingTask in
                let assigneeId = extracted.assigneeName.flatMap {
                    NameMatcher.matchAssignee(name: $0, candidates: members)
                }

                let dueDate = extracted.dueDate.flatMap {
                    NameMatcher.parseDate(from: $0)
                }

                return MeetingTask(
                    meetingId: meetingId,
                    organizationId: organizationId,
                    title: extracted.title,
                    assigneeId: assigneeId,
                    dueDate: dueDate,
                    isCompleted: false,
                    createdAt: Date(),
                    extractedFrom: extracted.context
                )
            }

            // Save tasks to database
            if !tasks.isEmpty {
                _ = try await taskService.createTasks(tasks)
            }

            processingState = .completed

        } catch {
            processingState = .failed(error.localizedDescription)
        }
    }
}
```

```swift
actor TaskService {
    func createTasks(_ tasks: [MeetingTask]) async throws -> [MeetingTask] {
        let dtos = tasks.map { $0.toDTO() }

        let response: [TaskDTO] = try await supabase
            .from("tasks")
            .insert(dtos)
            .select()
            .execute()
            .value

        return response.map { MeetingTask(from: $0) }
    }

    func fetchTasksForMeeting(meetingId: UUID) async throws -> [MeetingTask] {
        let response: [TaskDTO] = try await supabase
            .from("tasks")
            .select()
            .eq("meeting_id", value: meetingId.uuidString)
            .order("created_at", ascending: true)
            .execute()
            .value

        return response.map { MeetingTask(from: $0) }
    }

    func fetchTasksForOrganization(organizationId: UUID) async throws -> [MeetingTask] {
        let response: [TaskDTO] = try await supabase
            .from("tasks")
            .select()
            .eq("organization_id", value: organizationId.uuidString)
            .order("due_date", ascending: true)  // Soonest deadlines first
            .execute()
            .value

        return response.map { MeetingTask(from: $0) }
    }

    func updateTaskCompletion(taskId: UUID, isCompleted: Bool) async throws -> MeetingTask {
        let dto = TaskDTO(isCompleted: isCompleted)  // Partial update

        let response: TaskDTO = try await supabase
            .from("tasks")
            .update(dto)
            .eq("id", value: taskId.uuidString)
            .select()
            .single()
            .execute()
            .value

        return MeetingTask(from: response)
    }

    func deleteTask(taskId: UUID) async throws {
        try await supabase
            .from("tasks")
            .delete()
            .eq("id", value: taskId.uuidString)
            .execute()
    }
}
```

## Testing Strategy

**Manual testing:**
1. Upload meeting document with clear action items (e.g., "John will prepare budget by Friday")
2. Verify tasks extracted and saved to database
3. Check assignee name matching:
   - Exact name: "John Smith" → matches User(name: "John Smith")
   - Partial name: "John" → matches User(name: "John Smith")
   - Email: "john@example.com" → matches User(email: "john@example.com")
4. Verify due date parsing:
   - ISO8601: "2026-01-25" → Date(2026-01-25)
   - Natural language: "next Friday" → correct future date
5. Test with document containing no tasks → verify graceful handling

**Sample test documents:**
- "Alice will review designs by Monday. Bob to prepare presentation for next meeting."
- "TODO: Update documentation (Sarah), Deploy to production (urgent, Tom)"
- "Follow up with client - no specific assignee mentioned"

**Edge cases:**
- Typo in name: "Jhon" → should still match "John"
- Ambiguous name: "Smith" when two members have last name Smith → no assignee
- Invalid date: "sometime next month" → due date null
- No tasks in document → return empty array, don't fail

## Notes

**Decisions:**
- Structured outputs guarantee 100% JSON schema compliance (vs <40% with unstructured)
- Fuzzy name matching with confidence threshold (exact > email > partial > none)
- Priority inferred from urgency language in document
- Tasks without assignees still created (can be assigned manually later)
- Tasks table has ON DELETE CASCADE (deleting meeting deletes tasks)

**Name matching strategy:**
1. Exact full name match (highest confidence)
2. Email prefix match (high confidence)
3. Partial name component match (medium confidence)
4. No match → assigneeId null (user can assign manually)

**Limitations:**
- No NLP for complex date parsing (only common patterns)
- No position tracking in this plan (can add in Phase 7 polish)
- No duplicate task detection (AI might extract same task twice)

**Future considerations (not in this plan):**
- Machine learning for better name matching (training on org data)
- Natural language date parsing with DateComponents parser
- Duplicate task detection and merging
- Task priority adjustment based on meeting importance
- Position/role extraction ("John - President" → store role)

**Dependencies for next plan:**
- Plan 05-05 will display extracted tasks in meeting detail view
- Phase 6 will create dedicated task list view with completion tracking
